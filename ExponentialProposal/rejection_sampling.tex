\section*{Problem Statement}
We aim to perform rejection sampling to draw samples from a target distribution $f(x)$ using an instrumental (proposal) distribution $g(x)$. 

\begin{itemize}
    \item \textbf{Target:} $f(x) = \text{Gamma}(2,1) = \frac{x e^{-x}}{6e^{-5}}$ (normalized over the relevant interval)
    \item \textbf{Proposal:} $g(x) = \text{Exp}(0.8) = 0.8 e^{-0.8x}$
\end{itemize}

\section*{1. Finding the Ratio of Target to Proposal}
To find the scaling factor $M$ such that $f(x) \leq M g(x)$ for all $x$, we first calculate the ratio:

\begin{align*}
\frac{f(x)}{g(x)} &= \frac{\frac{x e^{-x}}{6e^{-5}}}{\frac{4}{5} e^{-\frac{4}{5}x}} \\
&= \frac{1}{6e^{-5}} \left( \frac{x e^{-x}}{\frac{4}{5} e^{-\frac{4}{5}x}} \right) \\
&= \frac{5}{24e^{-5}} (x) e^{-x - (-\frac{4}{5}x)} \\
&= \frac{5}{24e^{-5}} (x) e^{-\frac{1}{5}x}
\end{align*}

Thus, our ratio function is:
\[ \frac{f(x)}{g(x)} = \frac{5x e^{-\frac{1}{5}x}}{24e^{-5}} \]

\section*{2. Maximizing the Ratio}
We find the value of $x$ that maximizes this ratio by taking the derivative with respect to $x$ and setting it to zero. Using the product rule with $u = x$ and $v = e^{-\frac{1}{5}x}$:

\[ \frac{d}{dx} \left[ \frac{f(x)}{g(x)} \right] = \frac{5}{24e^{-5}} \left[ \left( -\frac{1}{5} \right) x e^{-\frac{1}{5}x} + e^{-\frac{1}{5}x} \right] \]

Setting the derivative to zero:
\[ 0 = \frac{5}{24e^{-5}} e^{-\frac{1}{5}x} \left( -\frac{1}{5}x + 1 \right) \]

Since $\frac{5}{24e^{-5}} e^{-\frac{1}{5}x} \neq 0$ for all finite $x$, we solve:
\begin{align*}
-\frac{1}{5}x + 1 &= 0 \\
-\frac{1}{5}x &= -1 \\
x &= 5
\end{align*}

\section*{3. Calculating the Scaling Factor $M$}
The maximum ratio occurs at $x = 5$. We calculate $M$ by evaluating the ratio at this point:

\begin{align*}
M &= \frac{f(5)}{g(5)} \\
&= \frac{\frac{5e^{-5}}{6e^{-5}}}{\frac{4}{5}e^{-\frac{4}{5}(5)}} \\
&= \frac{5/6}{4/5 e^{-4}} \\
M &\approx 56.873
\end{align*}
